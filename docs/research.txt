-- ============================================================
-- SECTION 4: Does a retail-shaped C_TradeSkillUI namespace exist,
-- and if so what does it actually expose? (dumps every function
-- name it finds - this is the single most useful command here)
-- ============================================================

/run if C_TradeSkillUI then local n=0; for k in pairs(C_TradeSkillUI) do print("C_TradeSkillUI."..k); n=n+1 end; print(n.." total") else print("C_TradeSkillUI does not exist") end

-- 4b. If 4 found GetRecipeInfo, try it against a known recipe spell ID (recipeID often == spellID in this data model)
/run if C_TradeSkillUI and C_TradeSkillUI.GetRecipeInfo then local info = C_TradeSkillUI.GetRecipeInfo(1229745); if info then for k,v in pairs(info) do print(k, tostring(v)) end else print("GetRecipeInfo(1229745) returned nil") end end

-- 4c. If 4 found GetRecipeSchematic (the modern reagent/output API), try it too
/run if C_TradeSkillUI and C_TradeSkillUI.GetRecipeSchematic then local s = C_TradeSkillUI.GetRecipeSchematic(1229745, false); if s then print("got schematic, reagentSlotCount="..tostring(s.reagentSlotCount)) else print("GetRecipeSchematic(1229745) returned nil") end end

-- 4d. Old classic-shape equivalent, for comparison
/run print("GetTradeSkillInfo exists:", GetTradeSkillInfo ~= nil, " GetTradeSkillReagentInfo exists:", GetTradeSkillReagentInfo ~= nil, " DoTradeSkill exists:", DoTradeSkill ~= nil)

-- ============================================================
-- SECTION 5: Resolve the First Aid Kit mystery - enumerate the
-- First Aid tab of your spellbook directly to find the real
-- craft-recipe spell (works whether or not professions use the
-- classic skill-line list or the retail GetProfessions() shape)
-- ============================================================

-- 5a. List every spellbook tab (profession tabs show up here on both classic and pre-C_SpellBook retail)
/run for i=1,GetNumSpellTabs() do local name,_,offset,numSpells = GetSpellTabInfo(i); print(i, name, "offset="..offset, "numSpells="..numSpells) end

-- 5b. If 5a shows a "First Aid" tab, plug its offset/numSpells in here to list every spell in it
/run local offset, numSpells = OFFSET, NUMSPELLS; for i=offset+1, offset+numSpells do local name = GetSpellBookItemName(i, "spell"); if name then print(i, name) end end

-- 5c. Modern-retail equivalent, in case the old spellbook API above returns nothing
/run if C_SpellBook then print("C_SpellBook exists"); if C_SpellBook.GetNumSpellBookSkillLines then print("numSkillLines="..C_SpellBook.GetNumSpellBookSkillLines()) end else print("no C_SpellBook") end
